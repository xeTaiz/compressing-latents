# Install with --no-deps into a separate target (never into or over the CUDA PyTorch env):
#   python -m pip install --no-cache-dir --no-deps --target /dev/shm/compressing-latents-deps \
#     -r requirements-extraction-worker.txt
#   PYTHONPATH=/dev/shm/compressing-latents-deps:$PYTHONPATH python -m compression.extract ...
# The base runtime supplies torch/torchvision (CUDA), numpy, Pillow, einops and opencv-python-headless.
